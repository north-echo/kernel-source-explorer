cmake_minimum_required(VERSION 3.20)
project(kse_indexer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
if(LLVM_VERSION_MAJOR LESS 22 OR LLVM_VERSION_MAJOR GREATER 23)
  message(FATAL_ERROR "Supported LLVM majors: 22 through 23; use matching Clang libraries")
endif()
add_executable(kse-index main.cpp)
target_include_directories(kse-index SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_compile_definitions(kse-index PRIVATE ${LLVM_DEFINITIONS})
target_link_libraries(kse-index PRIVATE clang-cpp LLVM)
